/**
 * Audit Kinerja Kabinet Samasta: penerima isian kementerian dan penyimpan hasil audit.
 * Tempel seluruh kode ini di Google Sheets BARU: Ekstensi > Apps Script. Panduan: PANDUAN-AUDIT.txt
 */
var KODE_AKSES = '';     // WAJIB DIISI: kode rahasia admin, minimal 6 karakter. Sama dengan kode yang diketik di halaman admin.
var SHEET_ID = '';       // OPSIONAL. Isi ID Spreadsheet bila skrip dibuat terpisah dari Sheet.
var MAX_DATA = 20000;    // batas ukuran satu isian (karakter)

function setup() {       // jalankan sekali
  var ss = ss_();
  [['Kementerian', ['ID', 'Nama', 'Kepala', 'Token', 'Periode']],
   ['Isian', ['SID', 'Kementerian', 'Versi', 'Dibuat', 'Diubah', 'StatusAudit', 'Catatan', 'Skor', 'Data (JSON)']]
  ].forEach(function (t) {
    var sh = ss.getSheetByName(t[0]) || ss.insertSheet(t[0]);
    if (sh.getLastRow() === 0) { sh.appendRow(t[1]); sh.setFrozenRows(1); }
    sh.getRange(1, 1, sh.getMaxRows(), 2).setNumberFormat('@');
  });
}
function ss_() { return SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet(); }
function reply_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function doGet() { return reply_({ ok: true, sheet: ss_() ? 'terhubung' : 'tidak', kode: KODE_AKSES.length >= 6 ? 'sudah diatur' : 'BELUM diatur' }); }
function s_(v, n) { return String(v == null ? '' : v).replace(/[\u0000-\u001f]/g, ' ').slice(0, n); }
function admin_(d) {     // kode dikirim lewat POST; 10 kali salah = ditahan 10 menit
  var c = CacheService.getScriptCache(), f = Number(c.get('gagal') || 0);
  if (f >= 10) return 'tunggu';
  if (KODE_AKSES.length < 6) return 'belumkode';
  if (String(d.kode || '') !== KODE_AKSES) { c.put('gagal', String(f + 1), 600); return 'kode'; }
  c.remove('gagal'); return '';
}
function kem_(mid, tk) {  // cari kementerian + cek token
  var sh = ss_().getSheetByName('Kementerian'); if (!sh || sh.getLastRow() < 2) return null;
  var v = sh.getRange(2, 1, sh.getLastRow() - 1, 5).getValues();
  for (var i = 0; i < v.length; i++) if (String(v[i][0]) === String(mid)) return String(v[i][3]) === String(tk) && tk ? { nama: v[i][1], kepala: v[i][2], periode: v[i][4] } : null;
  return null;
}
function doPost(e) {
  try {
    var d = JSON.parse((e && e.postData && e.postData.contents) || '{}'), a = d.aksi;
    if (a === 'masuk' || a === 'sinkron' || a === 'ambil' || a === 'audit') {
      var er = admin_(d); if (er) return reply_({ ok: false, error: er });
      if (a === 'masuk') return reply_({ ok: true });
      var lock = LockService.getScriptLock(); lock.waitLock(20000);
      try { return reply_(a === 'sinkron' ? sinkron_(d) : a === 'ambil' ? ambil_() : audit_(d)); } finally { lock.releaseLock(); }
    }
    var k = kem_(d.mid, d.tk); if (!k) return reply_({ ok: false, error: 'tautan' });
    if (a === 'info') return reply_({ ok: true, nama: k.nama, kepala: k.kepala, periode: k.periode });
    if (a === 'status') return reply_(status_(d));
    if (a === 'kirim') { var l2 = LockService.getScriptLock(); l2.waitLock(20000); try { return reply_(kirim_(d)); } finally { l2.releaseLock(); } }
    return reply_({ ok: false, error: 'aksi' });
  } catch (err) { console.error(err); return reply_({ ok: false, error: 'server' }); }
}
function sinkron_(d) {   // admin mengirim daftar kementerian + token
  var sh = ss_().getSheetByName('Kementerian'), L = d.daftar || [];
  if (!Array.isArray(L) || L.length > 60) return { ok: false, error: 'data' };
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 5).clearContent();
  sh.getRange(1, 1, Math.max(sh.getMaxRows(), L.length + 1), 5).setNumberFormat('@');
  if (L.length) sh.getRange(2, 1, L.length, 5).setValues(L.map(function (m) { return [s_(m.id, 40), s_(m.name, 120), s_(m.head, 120), s_(m.tk, 40), s_(m.periode, 20)]; }));
  return { ok: true };
}
function ambil_() {
  var sh = ss_().getSheetByName('Isian'), out = [];
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 9).getValues().forEach(function (r) {
    var dt; try { dt = JSON.parse(r[8]); } catch (x) { dt = {}; }
    out.push({ sid: r[0], mid: r[1], ver: r[2], at: r[3], upd: r[4], aud: r[5], cat: r[6], skor: r[7], data: dt });
  });
  return { ok: true, data: out };
}
function audit_(d) {     // admin menyimpan status audit + catatan
  var sh = ss_().getSheetByName('Isian'), st = ['menunggu', 'revisi', 'sah'].indexOf(d.aud) >= 0 ? d.aud : 'menunggu';
  if (sh.getLastRow() < 2) return { ok: false, error: 'tidak ada' };
  var ids = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) if (ids[i][0] === d.sid) {
    sh.getRange(i + 2, 5).setValue(Date.now()); sh.getRange(i + 2, 6, 1, 3).setValues([[st, s_(d.cat, 1000), Number(d.skor) || '']]);
    return { ok: true };
  }
  return { ok: false, error: 'tidak ada' };
}
function status_(d) {    // kementerian melihat status isiannya sendiri
  var sh = ss_().getSheetByName('Isian'), want = (d.sids || []).slice(0, 50), out = [];
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 9).getValues().forEach(function (r) {
    if (want.indexOf(r[0]) < 0 || String(r[1]) !== String(d.mid)) return;
    var o = { sid: r[0], aud: r[5], cat: r[6], ver: r[2] };
    if (r[5] === 'revisi') { try { o.data = JSON.parse(r[8]); } catch (x) {} }
    out.push(o);
  });
  return { ok: true, data: out };
}
function kirim_(d) {
  var x = d.data || {}, js = JSON.stringify(x);
  if (js.length > MAX_DATA || !x.name || !x.obj) return { ok: false, error: 'data' };
  if (!Array.isArray(x.kpis) || !x.kpis.length || x.kpis.length > 20) return { ok: false, error: 'kpi' };
  var tw = 0; x.kpis.forEach(function (k) { tw += Number(k.w) || 0; });
  if (tw !== 100) return { ok: false, error: 'bobot' };
  var sh = ss_().getSheetByName('Isian'), now = Date.now(), json = JSON.stringify(x);
  if (d.sid) {           // perbaikan: hanya boleh bila belum disahkan dan milik kementerian yang sama
    if (sh.getLastRow() > 1) {
      var v = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
      for (var i = 0; i < v.length; i++) if (v[i][0] === d.sid) {
        if (String(v[i][1]) !== String(d.mid) || v[i][5] === 'sah') return { ok: false, error: 'terkunci' };
        sh.getRange(i + 2, 3, 1, 4).setValues([[Number(v[i][2]) + 1, v[i][3], now, 'menunggu']]); sh.getRange(i + 2, 9).setValue(json);
        return { ok: true, sid: d.sid };
      }
    }
    return { ok: false, error: 'tidak ada' };
  }
  var sid = 'S' + now.toString(36) + Math.random().toString(36).slice(2, 6);
  sh.appendRow([sid, String(d.mid), 1, now, now, 'menunggu', '', '', json]);
  return { ok: true, sid: sid };
}
